! Aplicar este bloque en el switch Jueces
enable
configure terminal
hostname Jueces
vlan 20
 name JUECES
interface range FastEthernet0/1 - 24
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
interface range GigabitEthernet0/1 - 2
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
end
write memory
